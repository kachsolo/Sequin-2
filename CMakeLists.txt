cmake_minimum_required(VERSION 3.22)
project(Sequin2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(Sequin2 COMPANY_NAME "Easy Audio" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE PLUGIN_MANUFACTURER_CODE SqAu PLUGIN_CODE SqV2 FORMATS VST3 PRODUCT_NAME "Sequin 2 Prototype")
juce_generate_juce_header(Sequin2)
target_sources(Sequin2 PRIVATE Source/Plugin.cpp Source/HarmonicEngine.h)
target_compile_definitions(Sequin2 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Sequin2 PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
